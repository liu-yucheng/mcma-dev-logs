\documentclass[border=8pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=13cm, height=9.5cm,
    xlabel=$x$, ylabel=$y$, zlabel=$z$,
    xmin=-2.6, xmax=2.6, ymin=-1.8, ymax=6.6, zmin=-2.6, zmax=2.6,
    view={25}{18},
    title={Solid enclosed by $x^2+z^2=4$, $y=-1$ and $y=4-z$}]
    \addplot3[surf, domain=0:360, samples=49, y domain=0:1, samples y=6,
              opacity=0.28, blue!70!black, shader=flat, line width=0.1pt]
        ({2*cos(x)}, {-1 + y*(5 - 2*sin(x))}, {2*sin(x)});
    \addplot3[surf, domain=0:360, samples=49, y domain=0:2, samples y=10,
              opacity=0.85, green!55!black!55, shader=flat, line width=0.1pt]
        ({y*cos(x)}, {-1}, {y*sin(x)});
    \addplot3[surf, domain=0:360, samples=49, y domain=0:2, samples y=10,
              opacity=0.85, orange!75!red!60, shader=flat, line width=0.1pt]
        ({y*cos(x)}, {4 - y*sin(x)}, {y*sin(x)});
    \addplot3[very thick, black, domain=0:360, samples=97] ({2*cos(x)}, {-1}, {2*sin(x)});
    \addplot3[very thick, black, domain=0:360, samples=97] ({2*cos(x)}, {4 - 2*sin(x)}, {2*sin(x)});
\end{axis}
\end{tikzpicture}
\end{document}
