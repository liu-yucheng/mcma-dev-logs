\documentclass[border=10pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=10.5cm,height=9cm,
  view={-58}{20},
  xlabel={$x$},ylabel={$y$},zlabel={$z$},
  xmin=-1,xmax=1,ymin=-1,ymax=1,zmin=-1,zmax=1,
  xtick={-1,0,1},ytick={-1,0,1},ztick={-1,0,1},
  tick label style={font=\tiny},
  grid=major,grid style={gray!20},
  title={\footnotesize (b) Sphere $\rho=a$ and cylinder $r=a\cos\theta$ on the same screen ($a=1$)}]

  % ---------------- sphere  rho = a ----------------
  \addplot3[surf, blue!45!gray, opacity=0.10,
             domain=0:360, y domain=0:180, samples=20, samples y=12]
    ({sin(y)*cos(x)},{sin(y)*sin(x)},{cos(y)});
  \addplot3[blue!55!black, thin, opacity=0.8, domain=0:180, samples=40]
    ({sin(x)},{0},{cos(x)});
  \addplot3[blue!55!black, thin, opacity=0.8, domain=0:180, samples=40]
    ({0},{sin(x)},{cos(x)});
  \addplot3[blue!55!black, thin, opacity=0.8, domain=0:360, samples=60]
    ({0.866*cos(x)},{0.866*sin(x)},{0.5});
  \addplot3[blue!55!black, thin, opacity=0.8, domain=0:360, samples=60]
    ({0.866*cos(x)},{0.866*sin(x)},{-0.5});

  % ---------------- the solid: spherical caps  z = +- sqrt(a^2 - r^2) ----------------
  \addplot3[surf, green!45!black, opacity=0.85,
             domain=-90:90, y domain=0:1, samples=20, samples y=10]
    ({y*cos(x)^2},{y*cos(x)*sin(x)},{sqrt(1-(y*cos(x))^2)});
  \addplot3[surf, green!45!black, opacity=0.85,
             domain=-90:90, y domain=0:1, samples=20, samples y=10]
    ({y*cos(x)^2},{y*cos(x)*sin(x)},{-sqrt(1-(y*cos(x))^2)});

  % ---------------- cylinder  r = a cos(theta) ----------------
  \addplot3[surf, orange, opacity=0.16,
             domain=-90:90, y domain=-1:1, samples=20, samples y=2]
    ({cos(x)^2},{cos(x)*sin(x)},{y});
  \addplot3[orange!80!black, thin, opacity=0.9, domain=-90:90, samples=60]
    ({cos(x)^2},{cos(x)*sin(x)},{0});
  \addplot3[orange!80!black, thin, opacity=0.9, domain=-1:1, samples=2] ({1},{0},{x});
  \addplot3[orange!80!black, thin, opacity=0.9, domain=-1:1, samples=2] ({0.75},{0.433},{x});
  \addplot3[orange!80!black, thin, opacity=0.9, domain=-1:1, samples=2] ({0.75},{-0.433},{x});
  \addplot3[orange!80!black, thin, opacity=0.9, domain=-1:1, samples=2] ({0.25},{0.433},{x});
  \addplot3[orange!80!black, thin, opacity=0.9, domain=-1:1, samples=2] ({0.25},{-0.433},{x});

  % ---------------- Viviani intersection curves ----------------
  \addplot3[red!80!black, thick, domain=-90:90, samples=80]
    ({cos(x)^2},{cos(x)*sin(x)},{abs(sin(x))});
  \addplot3[red!80!black, thick, domain=-90:90, samples=80]
    ({cos(x)^2},{cos(x)*sin(x)},{-abs(sin(x))});

  % ---------------- labels ----------------
  \node[blue!55!black,font=\scriptsize] at (axis cs:-0.60,0,0.95) {sphere $\rho=a$};
  \node[orange!80!black,font=\scriptsize] at (axis cs:0.10,0.90,-0.95) {cylinder $r=a\cos\theta$};
  \node[green!40!black,font=\scriptsize] at (axis cs:-0.85,-0.55,0.30) {solid cut out};
  \node[red!80!black,font=\scriptsize] at (axis cs:0.95,0.45,0.90) {Viviani curve};
  \node[font=\scriptsize] at (axis cs:0.10,0.10,-0.16) {O};
\end{axis}
\end{tikzpicture}
\end{document}
