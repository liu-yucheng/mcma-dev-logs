\documentclass[border=10pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{colormaps}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12cm, height=12cm,
    view={-55}{20},
    xlabel={$x$}, ylabel={$y$}, zlabel={$z$},
    xmin=-1.15, xmax=1.15,
    ymin=-1.15, ymax=1.15,
    zmin=-1.15, zmax=1.15,
    xtick={-1,0,1}, ytick={-1,0,1}, ztick={-1,0,1},
    enlargelimits=false,
    axis lines=center,
    every axis x label/.style={at={(axis description cs:1.05,0.36)}, anchor=north},
    every axis y label/.style={at={(axis description cs:0.45,1.05)}, anchor=south},
    every axis z label/.style={at={(axis description cs:0.5,1.04)}, anchor=south},
    title={Solid cut from sphere by cylinder $r=a\cos\theta$},
    title style={align=center, font=\bfseries},
]

% Sphere wireframe: x^2+y^2+z^2 = a^2
% Parametric: u in [0,2pi], v in [0,pi]
\addplot3[
    mesh, black!55, very thin, samples=24, samples y=14,
    domain=0:360, y domain=0:180,
] (
    {cos(x)*sin(y)},
    {sin(x)*sin(y)},
    {cos(y)}
);
% Add a label for the sphere
\addplot3[only marks, mark=*, mark size=1.2pt, black] coordinates {(1,0,0)};

% Cylinder (x-a/2)^2 + y^2 = (a/2)^2, with z in [-1,1]
% Parametrize: theta in [-pi/2, pi/2], z in [-1,1]
\addplot3[
    surf, shader=interp, opacity=0.45,
    samples=24, samples y=10,
    domain=-90:90, y domain=-1:1,
] (
    {0.5 + 0.5*cos(x)},
    {0.5*sin(x)},
    {y}
);

% Top and bottom circle outlines of the cylinder (orange)
\addplot3[
    domain=-90:90, samples=80, thick, orange,
] (
    {0.5 + 0.5*cos(x)},
    {0.5*sin(x)},
    {1}
);
\addplot3[
    domain=-90:90, samples=80, thick, orange,
] (
    {0.5 + 0.5*cos(x)},
    {0.5*sin(x)},
    {-1}
);

% Sphere-cylinder intersection (red)
% r = cos(theta), z = |sin(theta)|; use param theta in [-pi/2, pi/2]
\addplot3[
    domain=-90:90, samples=120, very thick, red,
] (
    {cos(x)*cos(x)},          % r*cos(theta) = cos^2(theta)
    {cos(x)*sin(x)},          % r*sin(theta) = sin(theta)cos(theta)
    {abs(sin(x))}             % z = |sin(theta)|
);
\addplot3[
    domain=-90:90, samples=120, very thick, red,
] (
    {cos(x)*cos(x)},
    {cos(x)*sin(x)},
    {-abs(sin(x))}
);

% Axis labels (text)
\node at (axis cs:1.25,0,0) {\scriptsize$x$};
\node at (axis cs:0,1.25,0) {\scriptsize$y$};
\node at (axis cs:0,0,1.25) {\scriptsize$z$};

\end{axis}
\end{tikzpicture}
\end{document}
