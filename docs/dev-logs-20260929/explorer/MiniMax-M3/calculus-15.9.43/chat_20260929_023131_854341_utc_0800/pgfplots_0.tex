\documentclass[border=5pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=11cm, height=17cm,
    title={Silo: Cylinder ($r=3$, $h=10$) surmounted by Hemisphere ($r=3$)},
    xlabel={$x$}, ylabel={$y$}, zlabel={$z$},
    xmin=-4.5, xmax=4.5,
    ymin=-4.5, ymax=4.5,
    zmin=0, zmax=14,
    xtick={-3,-2,-1,0,1,2,3},
    ytick={-3,-2,-1,0,1,2,3},
    ztick={0,2,4,6,8,10,12},
    view={50}{22},
    axis lines=center,
    enlargelimits=false,
    grid=major,
    grid style={very thin, gray!25},
]

% Cylinder lateral surface (light gray, shaded)
\addplot3[
    surf, shader=faceted,
    domain=0:360, y domain=0:10,
    samples=50, samples y=20,
    fill=gray!25, draw=gray!55, faceted color=gray!55,
    z buffer=sort,
] ({3*cos(x)}, {3*sin(x)}, {y});

% Hemisphere (top) -- slightly lighter color to distinguish
\addplot3[
    surf, shader=faceted,
    domain=0:360, y domain=0:90,
    samples=50, samples y=18,
    fill=white, draw=gray!55, faceted color=gray!55,
    z buffer=sort,
] ({3*sin(y)*cos(x)}, {3*sin(y)*sin(x)}, {10+3*cos(y)});

% Outline rings for emphasis
% Bottom rim of cylinder
\addplot3[domain=0:360, samples=180, thick, black] ({3*cos(x)}, {3*sin(x)}, {0});
% Top rim of cylinder = bottom rim of hemisphere
\addplot3[domain=0:360, samples=180, thick, black] ({3*cos(x)}, {3*sin(x)}, {10});

% Two vertical edges on the front for clarity
\addplot3[black, thick] coordinates {(3,0,0) (3,0,10)};
\addplot3[black, thick] coordinates {(0,3,0) (0,3,10)};

% Meridian on hemisphere: from (3,0,10) up to apex (0,0,13)
\addplot3[domain=0:90, samples=40, thick, black]
    ({3*cos(y)}, {0}, {10+3*sin(y)});
\addplot3[domain=0:90, samples=40, thick, black]
    ({0}, {3*cos(y)}, {10+3*sin(y)});

\end{axis}
\end{tikzpicture}
\end{document}
