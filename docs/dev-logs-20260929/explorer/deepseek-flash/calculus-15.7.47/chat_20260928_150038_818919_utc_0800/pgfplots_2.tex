    xlabel={$y$}, ylabel={$z$},
    xmin=-0.05, xmax=1.15, ymin=-0.05, ymax=1.15,
    domain=0:1, samples=200,
    width=7cm, height=7cm,
    legend pos=north east,
    legend style={font=\small},
