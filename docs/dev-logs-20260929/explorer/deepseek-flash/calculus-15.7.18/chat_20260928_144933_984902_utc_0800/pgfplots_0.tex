\documentclass[border=10pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\pgfplotsset{trig format plots=rad}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    view={-62}{22},
    axis lines=center,
    xlabel=$x$, ylabel=$y$, zlabel=$z$,
    xmin=0, xmax=1.1,
    ymin=0, ymax=3.2,
    zmin=0, zmax=3.2,
    unit vector ratio=1 3 3,
    width=10cm, height=10cm,
    z buffer=sort,
]
% ---- stacked cross-sections D_x ----
\addplot3[surf, domain=0:3, y domain=0:1, samples=40, samples y=12,
          shader=flat, opacity=0.50, faceted color=black!35,
          color=blue!70!black]
    ({0}, {x}, {y*sqrt(9-x^2)});
\addplot3[surf, domain=0.6:3, y domain=0:1, samples=40, samples y=12,
          shader=flat, opacity=0.50, faceted color=black!35,
          color=blue!50!cyan]
    ({0.2}, {x}, {y*sqrt(9-x^2)});
\addplot3[surf, domain=1.2:3, y domain=0:1, samples=40, samples y=12,
          shader=flat, opacity=0.50, faceted color=black!35,
          color=cyan!80!black]
    ({0.4}, {x}, {y*sqrt(9-x^2)});
\addplot3[surf, domain=1.8:3, y domain=0:1, samples=40, samples y=12,
          shader=flat, opacity=0.50, faceted color=black!35,
          color=green!70!black]
    ({0.6}, {x}, {y*sqrt(9-x^2)});
\addplot3[surf, domain=2.4:3, y domain=0:1, samples=40, samples y=12,
          shader=flat, opacity=0.50, faceted color=black!35,
          color=yellow!70!black]
    ({0.8}, {x}, {y*sqrt(9-x^2)});

% highlighted typical slice at x = 0.5
\addplot3[surf, domain=1.5:3, y domain=0:1, samples=50, samples y=18,
          shader=flat, opacity=0.80, faceted color=red!50!black,
          color=red]
    ({0.5}, {x}, {y*sqrt(9-x^2)});

% ---- top curved edges of the slices ----
\addplot3[domain=0:3, samples=60, thick, black!80]
    ({0}, {x}, {sqrt(9-x^2)});
\addplot3[domain=0.6:3, samples=60, thick, black!80]
    ({0.2}, {x}, {sqrt(9-x^2)});
\addplot3[domain=1.2:3, samples=60, thick, black!80]
    ({0.4}, {x}, {sqrt(9-x^2)});
\addplot3[domain=1.8:3, samples=60, thick, black!80]
    ({0.6}, {x}, {sqrt(9-x^2)});
\addplot3[domain=2.4:3, samples=60, thick, black!80]
    ({0.8}, {x}, {sqrt(9-x^2)});
\addplot3[domain=1.5:3, samples=60, very thick, red!70!black]
    ({0.5}, {x}, {sqrt(9-x^2)});

% ---- vertical edges at y = 3x ----
\addplot3[domain=0:3, samples=2, thick, black!80]
    ({0}, {0}, {x});
\addplot3[domain=0:2.939, samples=2, thick, black!80]
    ({0.2}, {0.6}, {x});
\addplot3[domain=0:2.749, samples=2, thick, black!80]
    ({0.4}, {1.2}, {x});
\addplot3[domain=0:2.4, samples=2, thick, black!80]
    ({0.6}, {1.8}, {x});
\addplot3[domain=0:1.8, samples=2, thick, black!80]
    ({0.8}, {2.4}, {x});
\addplot3[domain=0:2.598, samples=2, thick, red!70!black]
    ({0.5}, {1.5}, {x});

% ---- boundary curves of the whole solid ----
\addplot3[domain=0:1.5708, samples=120, very thick, blue]
    ({cos(x)}, {3*cos(x)}, {3*sin(x)});
\addplot3[domain=0:1, samples=2, very thick, black]
    ({x}, {3}, {0});
\addplot3[domain=0:1, samples=2, very thick, purple]
    ({x}, {3*x}, {0});
\addplot3[domain=0:1.5708, samples=120, thick, green!60!black]
    ({0}, {3*cos(x)}, {3*sin(x)});
\end{axis}
\end{tikzpicture}
\end{document}
